function importFile(fpath){
	var Paths = Java.type('java.nio.file.Paths');
	var Files = Java.type('java.nio.file.Files');
	var lines = Files.readAllLines(Paths.get(fpath)
	             , Java.type('java.nio.charset.StandardCharsets').UTF_8);
	for (lnoif=0;lnoif<lines.length;lnoif++)
 	{	line=lines[lnoif]
 		values= line.split(";");
 		for (iif=0;iif<values.length;iif++)
 			{values[iif]=trimapo(values[iif]);
 			}
 		if (lnoif==0){
 			relModel=values[0];
 			//message(lnoif+': '+"RelModel= |"+relModel+"|");
 		} else 
 		if (lnoif==1){
 			// skip headers
 			//message(lnoif+': '+"headers= "+values);
 		} else {
	 		//message(lnoif+': '+values);
 			do1line(relModel,values);
 			lineanz++
 		} 		
 	};
}

function getColMap(colID,cmap){
	colsmap=[]
	if (cmap == null) return colsmap
	for (icolmap=0;icolmap<cmap.length;icolmap++){
		//message(colID+'|'+cmap[icolmap].getRelID())
		if (colID == cmap[icolmap].getRelID()) colsmap.push(cmap[icolmap])
	}
	return colsmap
}
function writeMapLine(fw,tab,enti,col,attr){

}
function printColsMap(fw,tab,lastOne,enti,cmap){
	cols=tab.getElementsCollection().toArray()
	//message(tab+'|'+enti+'|'+cols.length+'|'+cmap.length)
	if (enti!=null) writeMapLine(fw,tab,enti,null,null)
	//cols.sort()
	for (icpcm=0;icpcm<cols.length;icpcm++){
		col = cols[icpcm]
		colkey = tab+'.'+col
		colmaps=getColMap(col.getObjectID(),cmap)
		if (colmaps.length==0) {
			//nicht gemappte Columns werden nur 1x geschrieben.
			if (lastOne && !(colkey in writtenCols)) writeMapLine(fw,tab,null,col,null)
		}else{
			for (impcm=0;impcm<colmaps.length;impcm++){
				attrName=enti.getElementByID(colmaps[impcm].getLogID())
				//message('vor dem Schreiben='+colmaps[impcm].getLogID()+'|'+col+'|'+attrName)
				writeMapLine(fw,tab,enti,col, attrName)
				writtenCols[colkey]=true
			}
		}
	}
}
function getTabMap(modelMapping,tabID){
	maps=[]
	for (imtm=0;imtm<modelMapping.length;imtm++){
		if (tabID == modelMapping[imtm].getRelID()) maps.push(modelMapping[imtm])
	}
	return maps
}
function printCols(modelMapping,fw,tab){
	maps = getTabMap(modelMapping,tab.getObjectID())
	if (maps.length==0) {
		printColsMap(fw,tab,null,null)
	} else {
		for (impc=0;impc<maps.length;impc++){
			enti=logicalModel.getEntitySet().getObjectByID(maps[impc].getLogID())
			//message(enti+':'+maps[im].getLogType()+':'+maps[impc].getRelType())
			cmap=maps[impc].getContainedMappings().toArray()
			printColsMap(fw,tab,(impc==maps.length-1),enti,cmap)
		}
	}
}

function headers(rmodels){
	retval = ''
	for (ih=0;ih<rmodels.length;ih++){
		retval += ';'+rmodels[ih]
	}
	return retval
}

function getColName(rmod,tabID,colID){
	tab=rmod.getTableSet().getObjectByID(tabID)
	col = tab.getElementByID(colID)
	return tab+'.'+col
}
	
function	getMappings(rmodels){
	lmappings = {}
	if (rmodels == null) return lmappings
	for (imod=0;imod<rmodels.length;imod++){
		rmod=rmodels[imod]
		lmappings[rmod.getName()]={}
		//message(lmappings[rmod.getName()]['123:123'])
		modelMap= rmod.getRMExtendedMap().getContainerMappings().toArray();
		if (modelMap != null) {
			for (imgm=0;imgm<modelMap.length;imgm++){
				entiID = modelMap[imgm].getLogID()
				// Attributmappings
				cmap=modelMap[imgm].getContainedMappings().toArray()
				if (cmap != null) {
					for (icgm=0;icgm<cmap.length;icgm++){
						attrID = cmap[icgm].getLogID()
						key = entiID+'.'+attrID
						lmappings[rmod.getName()][key]=
						   getColName(rmod,modelMap[imgm].getRelID()
						            ,cmap[icgm].getRelID()) 
					}
				}
			}
		}
	}
	//message('vorReturn'+lmappings['Relational_1'])
	return lmappings
}

function printMatrix(fw,nurX) {
	logicalModel = model.getDesign().getLogicalDesign();
	entities= logicalModel.getEntitySet().toArray()
	rmodels = model.getDesign().getRelationalDesigns().toArray();
	mappings = getMappings(rmodels)
	if (entities != null) {
		fw.write('entity;attr'+headers(rmodels)+'\n')

		for (ientiac=0;ientiac<entities.length;ientiac++) {
			enti = entities[ientiac]
			entiID = enti.getObjectID()
			//message(enti)
			attrs=enti.getElementsCollection().toArray()
			if (attrs != null){
				for (iattrac=0;iattrac<attrs.length;iattrac++){
					attr=attrs[iattrac]
					attrID = attr.getObjectID()
					key=entiID+'.'+attrID
					
					line= '"'+enti+'";"'+attr+'"'
					for (irmpm=0;irmpm<rmodels.length;irmpm++){
						x= key in mappings[rmodels[irmpm].getName()] 
								? nurX ? 'X': mappings[rmodels[irmpm].getName()][key]
								: nvl('') 
						line += ';"'+x+'"'
					}
					line+='\n'
					fw.write(line)
				}
			}
		}
	    fw.write('created by ODM, '+curDateStr()+'\n')
	}
}
