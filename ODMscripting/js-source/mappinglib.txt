function getAllMappings(rmodels){
	containerMappings = {}
	attrMappings = {}
	if (rmodels == null) return 
	for (imod=0;imod<rmodels.length;imod++){
		rmod=rmodels[imod]
		modname=rmod.getName()
		tables = rmod.getTableSet()
		modelMap= rmod.getRMExtendedMap().getContainerMappings().toArray();
		if (modelMap != null) {
			for (imgm=0;imgm<modelMap.length;imgm++){
				logMap = modelMap[imgm]
				relName = tables.getObjectByID(logMap.getRelID())
				switch (logMap.getLogType()) {
				case ENTITYPE: 
					logName = entitySet.getObjectByID(logMap.getLogID())
					break
				 case RELATYPE:
					logName=relationSet.getObjectByID(logMap.getLogID())
					break;
				 case INHERITTYPE:
					logName=inheritanceSet.getObjectByID(logMap.getLogID())
					break;
				default:
					logName = null
				}
				
				containerMappings[logMap.getID()] = 
				   {modelName : modname
				   ,logID : logMap.getLogID()
				   ,logName : logName
				   ,logType : logMap.getLogType()
					,relID : logMap.getRelID()
					,tableName : relName
					,relType : logMap.getRelType()
					,srcEntityID : logMap.getRealSourceEntityID()
					,trgEntityID : logMap.getRealTargetEntityID()
				   }

				containedMaps = logMap.getContainedMappings()
				for (icm=0;icm<containedMaps.length;icm++) {
					cm = containedMaps[icm]
					attrMappings[cm.getID()] = 
					  {modelName : modname
					  ,parentID : logMap.getID()
					  ,logID : cm.getLogID()
					  ,logType : cm.getLogType()
					  ,relID : cm.getRelID()
					  ,relType : cm.getRelType()
					  }
				}
			}
		}
	}
}

function trimapo(s){
	return s.replace(/^"|"$/g, '');
}

function curDateStr(){
		var currentdate = new Date();
		return  currentdate.getDate() 
		+ "/" + (currentdate.getMonth()+1) 
		+ "/" + currentdate.getFullYear() + "  " 
		+ currentdate.getHours() + ":" 
		+ currentdate.getMinutes() + ":" + currentdate.getSeconds();
}

function getContainerMaps(tabID){
	maps=[]
	for (key in containerMappings) {
		if (tabID == containerMappings[key].relID) maps.push(containerMappings[key])
	}
	return maps
}

function getContainerMap(entiID,tabID){
	for (key in containerMappings) {
		if (tabID == containerMappings[key].relID && entiID == containerMappings[key].logID) return containerMappings[key]
	}
	return null
}

function getLogMap(logID,modName){
	maps=[]
	for (key in containerMappings) {
		if ((logID == containerMappings[key].logID)
		     && (modName == containerMappings[key].modelName)) maps.push(containerMappings[key])
	}
	return maps
}

function getColMaps(colID){
	colsmap=[]
	for (key in attrMappings){
		//message(colID+'|'+cmap[icolmap].getRelID())
		if (colID == attrMappings[key].relID) colsmap.push(attrMappings[key])

}
function getColMap(colID,attrID){
	for (key in attrMappings){
		if (attrMappings[key].modelName == "--debug--"){
		   message(key+'\n'+colID+'\n'+attrID+'\n'+attrMappings[key].relID+'\n'+attrMappings[key].logID)
		}
		if (colID == attrMappings[key].relID && attrID == attrMappings[key].logID) return attrMappings[key]
	}
	return null
}
function getAttrMap(attrID,modName){
	attrmaps=[]
	for (key in attrMappings){
		//message(colID+'|'+cmap[icolmap].getRelID())
		if ((attrID == attrMappings[key].logID)
		&& (modName == attrMappings[key].modelName)) attrmaps.push(attrMappings[key])
	}
	return attrmaps
}


function getRelationAttribute(attrCollection,attrName){
	for (grakey in attrCollection) {
		if (attrCollection[grakey].getName() == attrName) return attrCollection[grakey]
	}
	return null 
}
function getRelationAttrByID(attrCollection,attrID){
	for (grakey in attrCollection) {
		if (attrCollection[grakey].getObjectID() == attrID) return attrCollection[grakey]
	}
	return null 
}


function do1line(values){
	modelMapping = rmodel.getRMExtendedMap().getContainerMappings().toArray();
	elemName = values[0];
	attrName = values[1];
	tabName = values[2];
	colName = values[3];
	//message(entiName+' | '+attrName+' | '+tabName+' | '+colName)
	relation = null
	entity = entitySet.getByName(elemName);
	if (entity != null) {
		attribute=entity.getElementByName(attrName);
		//if (attribute = null){
			//entity.createAttribute(attrName)
			//attribute = entity.getElementByName(attrName);
			//attribute.setComment("Created on "+curDateStr() +" during import from "+imppath)		
		//}
	} else {// assume Relation 
		relation = relationSet.getByName(elemName)
		if (relation != null) {
			attribute=getRelationAttribute(relation.getAttributesCollection(),attrName);
		}
	}
	table = rmodel.getTableSet().getByName(tabName);	
	// Neue Tabellen werden erzeugt
	if ((table == null) && (tabName != "")) {
		rmodel.createTable(tabName);
		table = rmodel.getTableSet().getByName(tabName);	
		table.setComment("Created on "+curDateStr() +" during import from "+imppath)
		tabanz++
	}
	if (table != null){
		column=table.getElementByName(colName);
	} else  {
		column = null
	}
	// neue Columns werden erzeugt
	if ((column == null) && (colName != "")){
		//message('neue Column')
		table.createColumn(colName)
		column=table.getElementByName(colName);
		column.setComment("Created on "+curDateStr() +" during import from "+imppath)		
		colanz++
	}	
	if ((entity != null) && (table != null)){
		mapped = getContainerMap(entity.getObjectID(),table.getObjectID())
		if (mapped == null) {
			rmodel.getRMExtendedMap().createMapping(entity,table);
			tabmapanz++
		}
		if ((attribute != null) && (column != null)){
			mapped = getColMap(column.getObjectID(),attribute.getObjectID())
			
			if (mapped == null) {
				rmodel.getRMExtendedMap().createMapping(attribute,column);
				colmapanz++
     		}
     	}
	}
	if ((relation != null) && (table != null)){
		mapped = getContainerMap(relation.getObjectID(),table.getObjectID())
		if (mapped == null) {
			rmodel.getRMExtendedMap().createMapping(relation,table);
			tabmapanz++
		}
		if ((attribute != null) && (column != null)){
			mapped = getColMap(column.getObjectID(),attribute.getObjectID())
			if (mapped == null) {
				rmodel.getRMExtendedMap().createMapping(attribute,column);
				colmapanz++
     		}
     	}
	}
}

function headers(rmodels,nurX){
	retval = ''
	for (ih=0;ih<rmodels.length;ih++){
		retval += ';'+rmodels[ih]
		// extra header for table | attribute entries
		if (! nurX) { retval += ';column;ID'}
	}
	return retval
}

function getColName(rmod,tabID,colID){
	tab=rmod.getTableSet().getObjectByID(tabID)
	col = tab.getElementByID(colID)
	return tab+'.'+col
}

function colProperty(curmodel,tabname,colname,propName){
	table = curmodel.getTableSet().getByName(tabName);
	if (table == null) return ""
	column = table.getElementByName(colName);
	extID= column.getProperty(propName)
	return ((extID == null) || (extID == ".")) ? "" : extID
}

function findFK(fks,fkID){

function tabList (maps,elemType,curmodel) {
	if (maps == null) return ''
	retval = ''
	if (elemType != 'Entity') {
		fks = curmodel.getFKIndexAssociationSet().toArray()
	}
	for (i in maps) {
		retval += (retval == '') ? '' : EXCELSEPARATOR 
		if (elemType == 'Entity') {
			tabName = maps[i].tableName 
		} else {			
			fk = findFK(fks,maps[i].relID)
			if (fk != null) {
				tabName = fk.getTable()
			} else {
				tabName = maps[i].tableName // falls es m:n ist ist es eine Table
			}
		}
		retval += tabName
	}
	return retval
}

function elements(rmodels,logID,elemType,nurX){
	retval = ''
	for (ih=0;ih<rmodels.length;ih++){
		curmodel = rmodels[ih]
		retval += ';'
		maps = getLogMap(logID,curmodel)
		if (maps.length > 0) {
			retval += nurX ? 'X': tabList(maps,elemType,curmodel)
		}
		// Extra columns are empty
		retval += nurX ? '' : ';;'
	}
	return retval
}

function attrElements(rmodels,attrID,nurX) {
	retval = ''
	for (irmpm=0;irmpm<rmodels.length;irmpm++){
		curmodel = rmodels[irmpm]
		attrmaps = getAttrMap(attrID,curmodel.getName())
		//message(curmodel.getName()+':'+attrID+':'+attrmaps.length)
		if (attrmaps.length > 0) {
			if (nurX) { x = ';X' 
			} else {
				x = ''
				tabNames = ''
				colNames = ''
				colIDs = ''
				for (key in attrmaps){
					attrmap = attrmaps[key]
					tabName= containerMappings[attrmap.parentID].tableName
					colName = curmodel.getTableSet().getByName(tabName).getElementByID(attrmap.relID)
					tabNames += tabName + EXCELSEPARATOR
					colNames += colName + EXCELSEPARATOR
					colIDs += colProperty(curmodel,tabName,colName,EXTIDPROPNAME) + EXCELSEPARATOR
				}
				tabNames = tabNames.slice(0,-1)
				colNames = colNames.slice(0,-1)
				colIDs = colIDs.slice(0,-1)
				x +=  ';' + tabNames + ';' + colNames + ';' + colIDs  
			}
		} else {x = nurX ? ';' : ';;;'}
		retval += x
	}
	return retval
}

function logicalElement(rmodels,logElem,nurX){	
	elemID = logElem.getObjectID()
	attrs = null
	if (logElem.getObjectTypeName() == 'Entity'){
		line= logElem.getName()+';'
		line += elements(rmodels,logElem.getObjectID(),logElem.getObjectTypeName(),nurX)
		line += '\n'
		fw.write(line)
		attrs=logElem.getElementsCollection().toArray()
	} else if (logElem.getObjectTypeName() == 'Relation') {
		if (logElem.isManyToMany()) { // FK-Mapping ist nicht trivial bei seltsamen Modellen
			line= logElem.getName()+';'
			line += elements(rmodels,logElem.getObjectID(),logElem.getObjectTypeName(),nurX)
			line += '\n'
			fw.write(line)
			attrs=logElem.getAttributesCollection().toArray()	
		} 
	} else { // assume inheritance
		// Inheritance sind schon entities line += elements(rmodels,logElem.getObjectID(),logElem.getObjectTypeName(),nurX)
		// Inheritance becomes fk-columns
	}
 	if (attrs != null){
		for (iattrac=0;iattrac<attrs.length;iattrac++){
			attr = attrs[iattrac]
			attrID = attr.getObjectID()
			key=elemID+'.'+attrID

			line= logElem.getName()+';'+attr.getName()
			line += attrElements(rmodels,attrID,nurX)
			line += '\n'
			fw.write(line)
		}
	}
}
