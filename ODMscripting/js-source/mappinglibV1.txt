function getAllMappings(rmodels){
	containerMappings = []
	if (rmodels == null) return lmappings
	for (imod=0;imod<rmodels.length;imod++){
		rmod=rmodels[imod]
		modname=rmod.getName()
		modelMap= rmod.getRMExtendedMap().getContainerMappings().toArray();
		if (modelMap != null) {
			for (imgm=0;imgm<modelMap.length;imgm++){
				logMap = modelMap[imgm]
				containerMappings.push([modname,logMap.getLogID(),logMap.getLogType()
									,logMap.getRelID(),logMap.getRelType()
									,logMap.getRealSourceEntityID(),logMap.getRealTargetEntityID()])

				containedMaps = logMap.getContainedMappings()
				for (icm=0;icm<containedMaps.length;icm++) {
					cm = containedMaps[icm]
					attrMappings.push([modname,cm.getLogID(),cm.getLogType()
									,cm.getRelID(),cm.getRelType()])
				}
			}
		}
	}
}

}

function importFile(fpath){
	var Paths = Java.type('java.nio.file.Paths');
	var Files = Java.type('java.nio.file.Files');
	var lines = Files.readAllLines(Paths.get(fpath)
	             , Java.type('java.nio.charset.StandardCharsets').UTF_8);
	for (lnoif=0;lnoif<lines.length;lnoif++)
 	{	line=lines[lnoif]
 		values= line.split(";");
 		for (iif=0;iif<values.length;iif++)
 			{values[iif]=trimapo(values[iif]);
 			}
 		if (lnoif==0){
 			relModel=values[0];
 			// falls 1. Zeichen von Excel stammt, schneide es ab
 			 if (relModel.match(/^[^A-Za-z]/)) relModel = relModel.substring(1)
 			//message(lnoif+': '+"RelModel= |"+relModel+"|");
 		} else 
 		if (lnoif==1){
 			// skip headers
 			//message(lnoif+': '+"headers= "+values);
 		} else {
	 		//message(lnoif+': '+values);
 			do1line(relModel,values);
 			lineanz++
 		} 		
 	};
}

function nvl(x){return x==null ? "" : x }

function getColMap(colID,cmap){
	colsmap=[]
	if (cmap == null) return colsmap
	for (icolmap=0;icolmap<cmap.length;icolmap++){
		//message(colID+'|'+cmap[icolmap].getRelID())
		if (colID == cmap[icolmap].getRelID()) colsmap.push(cmap[icolmap])
	}

}
function writeMapLine(fw,tab,enti,col,attr){
	//message('write'+tab+enti+col+attr)
	fw.write(nvl(enti)+';'+nvl(attr)+';'+tab+';'+nvl(col)+'\n')
}
function printColsMap(fw,tab,lastOne,enti,cmap){
	cols=tab.getElementsCollection().toArray()
	//message(tab+'|'+enti+'|'+cols.length+'|'+cmap.length)
	if (enti != null) writeMapLine(fw,tab,enti,null,null)
	//cols.sort()
	for (icpcm=0;icpcm<cols.length;icpcm++){
		col = cols[icpcm]
		colkey = tab+'.'+col
		colmaps=getColMap(col.getObjectID(),cmap)
		if (colmaps.length==0) {
			//nicht gemappte Columns werden nur 1x geschrieben.
			if (lastOne && !(colkey in writtenCols)) writeMapLine(fw,tab,null,col,null)
		}else{
			if (enti != null) {
				for (impcm=0;impcm<colmaps.length;impcm++){
					attrName=enti.getElementByID(colmaps[impcm].getLogID())
					//message('vor dem Schreiben='+colmaps[impcm].getLogID()+'|'+col+'|'+attrName)
					writeMapLine(fw,tab,enti,col, attrName)
					writtenCols[colkey]=true
				}
			}
		}
	}
}
function getTabMap(modelMapping,tabID){
	maps=[]
	for (imtm=0;imtm<modelMapping.length;imtm++){
		if (tabID == modelMapping[imtm].getRelID()) maps.push(modelMapping[imtm])
	}
	return maps
}
function printCols(modelMapping,fw,tab){
	maps = getTabMap(modelMapping,tab.getObjectID())
	if (maps.length==0) {
		printColsMap(fw,tab,true,null,null)
	} else {
		for (impc=0;impc<maps.length;impc++){
			enti=logicalModel.getEntitySet().getObjectByID(maps[impc].getLogID())
			//message(enti+':'+maps[im].getLogType()+':'+maps[impc].getRelType())
			cmap=maps[impc].getContainedMappings().toArray()
			printColsMap(fw,tab,(impc==maps.length-1),enti,cmap)
		}
	}
}

function printCVS (modelMapping,fw){
	fw.write(relModel+';;;\n')

function headers(rmodels,nurX){
	retval = ''
	for (ih=0;ih<rmodels.length;ih++){
		retval += ';'+rmodels[ih]
		// extra header for table | attribute entries
		if (! nurX) { retval += ';column;ID'}
	}
	return retval
}

function getColName(rmod,tabID,colID){
	tab=rmod.getTableSet().getObjectByID(tabID)
	col = tab.getElementByID(colID)
	return tab+'.'+col
}
	
function	getMappings(rmodels){
	lmappings = {}
	if (rmodels == null) return lmappings
	for (imod=0;imod<rmodels.length;imod++){
		rmod=rmodels[imod]
		lmappings[rmod.getName()]={}
		//message(lmappings[rmod.getName()]['123:123'])
		modelMap= rmod.getRMExtendedMap().getContainerMappings().toArray();
		//message("modelMapAnz="+modelMap.length)
		if (modelMap != null) {
			for (imgm=0;imgm<modelMap.length;imgm++){
				entiID = modelMap[imgm].getLogID()
				// Attributmappings
				cmap=modelMap[imgm].getContainedMappings().toArray()
				if (cmap != null) {
					for (icgm=0;icgm<cmap.length;icgm++){
						//message("modelmap RELID="+cmap[imgm].getRelID()+"\nName="+cmap[icgm])};
						//rmod.getElementByID(modelMap[imgm].getRelID()))
						attrID = cmap[icgm].getLogID()
						key = entiID+'.'+attrID
						lmappings[rmod.getName()][key]=
						   getColName(rmod,modelMap[imgm].getRelID()
						            ,cmap[icgm].getRelID()) 
					}
				}
			}
		}
	}
	//message('vorReturn'+lmappings['Relational_1'])
	return lmappings
}
function colProperty(curmodel,tabname,colname,propName){
	table = curmodel.getTableSet().getByName(tabName);	
	column = table.getElementByName(colName);
	extID= column.getProperty(propName)
	return ((extID == null) || (extID == ".")) ? "" : extID
}

function logicalElement(mappings,rmodels,logElem,nurX){	
	elemID = logElem.getObjectID()
	//message(enti)
	if (logElem.getObjectTypeName() == 'Entity'){
		attrs=logElem.getElementsCollection().toArray()
	} else if (logElem.getObjectTypeName() == 'Relation') {
		attrs=logElem.getAttributesCollection().toArray()	
		if (! logElem.isManyToMany()) {
			// Relationships 1:1 and m:1 can become columns
			//srcCard = logElem.getSourceCardinalityString()
			//targCard = logElem.getTargetCardinalityString()
			//srcOpt = logElem.isOptionalSource()
			//targOpt = logElem.isOptionalTarget()
			//message(logElem.getName()+':'+srcCrd+':'+targCard+':'+srcOpt+':'+targOpt)
			line= logElem.getName()+';;'
			line+='\n'
			fw.write(line)
		}
	} else { // assume inheritance
		attrs=null
		line= logElem.getName()+';;'
		line+='\n'
		fw.write(line)
	}
 	if (attrs != null){
		for (iattrac=0;iattrac<attrs.length;iattrac++){
			attr=attrs[iattrac]
			attrID = attr.getObjectID()
			key=elemID+'.'+attrID

			line= logElem.getName()+';'+attr.getName()
			for (irmpm=0;irmpm<rmodels.length;irmpm++){
				curmodel = rmodels[irmpm]
				mapentry = mappings[curmodel.getName()][key]
				if (mapentry != null){
					mapentries = mapentry.split(".")
					tabName = mapentry.replace(/^([^\.]*)\.(.*)$/,'$1')
					colName = mapentry.replace(/^([^\.]*)\.(.*)$/,'$2')
					colID = colProperty(curmodel,tabName,colName,EXTIDPROPNAME)
					//message(logElem.getName()+' *2* '+attrs.length+'   '+mapentry+ '  '+irmpm+'   ')
					x =  nurX ? 'X': tabName + ';' + colName + ';' + colID 
				} else {x = nurX ? '' : ';;'}
				line += ';'+x
			}
			line+='\n'
			fw.write(line)
		}
	}
}

function printMatrix(fw,nurX) {
	logicalModel = model.getDesign().getLogicalDesign();
	rmodels = model.getDesign().getRelationalDesigns().toArray();
	mappings = getMappings(rmodels)	
	getAllMappings()
	//message("entities="+entities.length+"   Rels="+relations.length)

	entities= logicalModel.getEntitySet().toArray()
	if (entities != null) {
		fw.write('entity;attr'+headers(rmodels,nurX)+'\n')
		for (ientiac=0;ientiac<entities.length;ientiac++) {
			logicalElement(mappings,rmodels,entities[ientiac],nurX)
		}
	}
	relations = logicalModel.getRelationSet().toArray()
	if (relations != null) {
		for (irel=0;irel<relations.length;irel++) {
			logicalElement(mappings,rmodels,relations[irel],nurX)
		}
	}

	inheritrelations = logicalModel.getInheritanceRelationSet().toArray()
	if (inheritrelations != null) {
		for (irel=0;irel<inheritrelations.length;irel++) {
			logicalElement(mappings,rmodels,inheritrelations[irel],nurX)
		}
	}
	
	fw.write('created by ODM, '+curDateStr()+'\n')
}
