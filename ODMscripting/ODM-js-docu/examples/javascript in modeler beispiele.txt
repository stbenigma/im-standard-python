	function map(line,lm,rm) {
	    //rm.getAppView().logMessage(line);
	    //separate the names using the separator from the file
	  nparts = line.split(",");
	  //get names from logical model
	  lnames = nparts[0].split(".");
	  //get names from relational model - example assumes it's table_name.column_name - if schema is added then next processing need to be changed
	  rnames = nparts[1].split(".");
	  doMapping(lnames[0],lnames[1],rnames[0],rnames[1],lm,rm);
	}

	function doMapping(ename,aname,tname,cname,lm,rm){
	    entity = lm.getEntitySet().getByName(ename);
	    if(entity!=null){
	        //use getBySchemaAndName(String schema,String name) if schema is used
	        table = rm.getTableSet().getByName(tname);
	        if(table!=null){
	          rm.getRMExtendedMap().createMapping(entity,table);
	          attr = entity.getElementByName(aname);
	          col = table.getElementByName(cname);
	          if(attr!=null && col!=null){
	              rm.getRMExtendedMap().createMapping(attr,col);
	          }
	        }
	    }
	}
